# Plank dock-items seed: day-1 default for the user's dconf user-db.
#
# Loaded by sudhanix-firstlogin's sfl_seed_user_dconf() on first login,
# BEFORE Plank starts. Pre-populating user-db here is what suppresses
# Plank's first-run auto-detect from clobbering both the dconf default
# and the .dockitem files seeded from /etc/skel.
#
# Day-1 contents = the union of the default-enabled dock categories
# (see dock-essentials.txt and dock-office.txt). The Welcome panel's
# Customization tab can then add Code or Art categories on top, or
# remove Office, by rewriting this same key. Keep the order matching
# DOCK_CATEGORIES_FALLBACK in /usr/local/bin/sudhanix-welcome.
#
# prefix: /net/launchpad/plank/

[docks/dock1]
# Day-1 first-login state: empty AND auto-hidden. The welcome panel
# populates dock-items + flips hide-mode back to 'none' when the user
# clicks Apply or dismisses (default = all categories on dismiss). An
# explicit empty list (rather than absent) is what keeps Plank's auto-
# detect from clobbering the user-db on first run; see cttb-ansible#35.
# `@as []` is the GVariant type annotation forcing "array of strings"
# on an empty list -- dconf load can't infer the element type otherwise.
dock-items=@as []
hide-mode='auto'
